cmake_minimum_required(VERSION 3.21)
project(Misaka LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick OpenGL Network)
find_package(Qt6Keychain REQUIRED)
find_library(QRENCODE NAMES qrencode libqrencode.so.4 REQUIRED)
find_path(MPV_INCLUDE_DIR mpv/client.h REQUIRED)
find_library(MPV_LIBRARY NAMES mpv libmpv.so.2 REQUIRED)
add_library(player STATIC account.cpp video.cpp video.h)
target_include_directories(player PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE ${MPV_INCLUDE_DIR})
target_link_libraries(player PUBLIC Qt6::Quick Qt6::OpenGL Qt6::Network PRIVATE qt6keychain ${QRENCODE} ${MPV_LIBRARY})
add_executable(misaka main.cpp)
target_link_libraries(misaka PRIVATE player)
set(player_ui Main.qml Login.qml icons/play.svg icons/pause.svg icons/volume.svg icons/muted.svg)
qt_add_resources(misaka ui PREFIX "/" FILES ${player_ui})
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(playback_test tests/playback.cpp)
    target_link_libraries(playback_test PRIVATE player Qt6::Test)
    qt_add_resources(playback_test ui PREFIX "/" FILES ${player_ui})
    add_test(NAME playback COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixture.py $<TARGET_FILE:playback_test>)
endif()

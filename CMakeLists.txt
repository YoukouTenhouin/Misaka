cmake_minimum_required(VERSION 3.21)
project(Misaka LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick OpenGL)
find_path(MPV_INCLUDE_DIR mpv/client.h REQUIRED)
find_library(MPV_LIBRARY NAMES mpv libmpv.so.2 REQUIRED)
add_library(player STATIC video.cpp video.h)
target_include_directories(player PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE ${MPV_INCLUDE_DIR})
target_link_libraries(player PUBLIC Qt6::Quick Qt6::OpenGL PRIVATE ${MPV_LIBRARY})
add_executable(misaka main.cpp)
target_link_libraries(misaka PRIVATE player)
qt_add_resources(misaka ui PREFIX "/" FILES Main.qml)
